cmake_minimum_required(VERSION 3.24)
project(MinuvexConsumer LANGUAGES CXX)
option(MINUVEX_CONSUMER_NATIVE_HOST "Build the native Host sample as an external SDK consumer" OFF)
if(APPLE AND MINUVEX_CONSUMER_NATIVE_HOST)
    enable_language(OBJCXX)
endif()
if(NOT MINUVEX_SOURCE_DIR)
    message(FATAL_ERROR "Set MINUVEX_SOURCE_DIR to the Minuvex source checkout.")
endif()
set(MINUVEX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MINUVEX_BUILD_EDITOR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${MINUVEX_SOURCE_DIR}" engine)
add_executable(viewer_consumer Main.cpp)
target_link_libraries(viewer_consumer PRIVATE minuvex::viewer)
add_executable(document_consumer Document.cpp)
target_link_libraries(document_consumer PRIVATE minuvex::editor_document)
add_executable(animation_consumer Animation.cpp)
target_link_libraries(animation_consumer PRIVATE minuvex::scene_io)
add_executable(quality_consumer Quality.cpp)
target_link_libraries(quality_consumer PRIVATE minuvex::quality)
add_executable(telemetry_consumer Telemetry.cpp)
target_link_libraries(telemetry_consumer PRIVATE minuvex::telemetry)
add_executable(embedded_consumer Embedded.cpp)
target_link_libraries(embedded_consumer PRIVATE minuvex::embedded)
if(MINUVEX_CONSUMER_NATIVE_HOST)
    if(NOT MINUVEX_BUILD_RENDERER)
        message(FATAL_ERROR "The native Host consumer requires MINUVEX_BUILD_RENDERER=ON.")
    endif()
    if(WIN32)
        add_executable(embedded_host_consumer "${MINUVEX_SOURCE_DIR}/examples/EmbeddedViewer/MainD3D11.cpp")
        target_compile_definitions(embedded_host_consumer PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
        target_link_libraries(embedded_host_consumer PRIVATE minuvex::embedded d3d11 dxgi user32)
    elseif(APPLE)
        add_executable(embedded_host_consumer "${MINUVEX_SOURCE_DIR}/examples/EmbeddedViewer/MainMetal.mm")
        set_target_properties(embedded_host_consumer PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
        target_compile_options(embedded_host_consumer PRIVATE -fobjc-arc)
        target_link_libraries(embedded_host_consumer PRIVATE minuvex::embedded "-framework Cocoa" "-framework Metal" "-framework QuartzCore")
    endif()
    target_compile_definitions(embedded_host_consumer PRIVATE MINUVEX_EMBEDDED_SAMPLE="${MINUVEX_SOURCE_DIR}/examples/assets/AnimatedRibbon.gltf")
endif()
