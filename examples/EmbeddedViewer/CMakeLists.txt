if(WIN32)
    add_executable(minvex_embedded_host MainD3D11.cpp)
    target_compile_definitions(minvex_embedded_host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    target_link_libraries(minvex_embedded_host PRIVATE minvex::embedded d3d11 dxgi user32)
elseif(APPLE)
    add_executable(minvex_embedded_host MainMetal.mm)
    set_target_properties(minvex_embedded_host PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
    target_compile_options(minvex_embedded_host PRIVATE -fobjc-arc)
    target_link_libraries(minvex_embedded_host PRIVATE minvex::embedded "-framework Cocoa" "-framework Metal" "-framework QuartzCore")
endif()
set_target_properties(minvex_embedded_host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
target_compile_definitions(minvex_embedded_host PRIVATE MINVEX_EMBEDDED_SAMPLE="${PROJECT_SOURCE_DIR}/examples/assets/AnimatedRibbon.gltf")
minvex_configure_target(minvex_embedded_host)

    set_tests_properties(integration.embedded PROPERTIES TIMEOUT 45 LABELS "rendering;desktop;smoke;embedded")
