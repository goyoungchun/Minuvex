if(WIN32)
    add_executable(minuvex_embedded_host MainD3D11.cpp)
    target_compile_definitions(minuvex_embedded_host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    target_link_libraries(minuvex_embedded_host PRIVATE minuvex::embedded d3d11 dxgi user32)
elseif(APPLE)
    add_executable(minuvex_embedded_host MainMetal.mm)
    set_target_properties(minuvex_embedded_host PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
    target_compile_options(minuvex_embedded_host PRIVATE -fobjc-arc)
    target_link_libraries(minuvex_embedded_host PRIVATE minuvex::embedded "-framework Cocoa" "-framework Metal" "-framework QuartzCore")
endif()
set_target_properties(minuvex_embedded_host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
target_compile_definitions(minuvex_embedded_host PRIVATE MINUVEX_EMBEDDED_SAMPLE="${PROJECT_SOURCE_DIR}/examples/assets/AnimatedRibbon.gltf")
minuvex_configure_target(minuvex_embedded_host)
