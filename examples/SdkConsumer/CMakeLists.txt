cmake_minimum_required(VERSION 3.24)
project(MinvexSdkConsumer LANGUAGES CXX)
option(MINVEX_SDK_BUILD_VIEWER "Build native Viewer and rendering examples" OFF)
option(MINVEX_SDK_BUILD_EMBEDDED_HOST "Build a native Embedded Host example" OFF)
find_package(Minvex 1.16 CONFIG REQUIRED COMPONENTS core assets scene_io quality telemetry debug optimization editor_document viewer embedded)
if(MINVEX_SDK_BUILD_VIEWER OR MINVEX_SDK_BUILD_EMBEDDED_HOST)
    find_package(Minvex 1.16 CONFIG REQUIRED COMPONENTS graphics)
endif()
add_executable(minvex_sdk_cpu Cpu.cpp)
target_link_libraries(minvex_sdk_cpu PRIVATE minvex::editor_document minvex::viewer minvex::embedded)
add_executable(minvex_sdk_animation Animation.cpp)
target_link_libraries(minvex_sdk_animation PRIVATE minvex::scene_io)

# Compile every public header independently, including the generated Version.h.
get_target_property(_sdk_includes minvex::core INTERFACE_INCLUDE_DIRECTORIES)
foreach(_include IN LISTS _sdk_includes)
    file(GLOB _headers "${_include}/minvex/*.h")
    foreach(_header IN LISTS _headers)
        get_filename_component(_name "${_header}" NAME)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/headers/${_name}.cpp" "#include <minvex/${_name}>\n")
        list(APPEND _header_checks "${CMAKE_CURRENT_BINARY_DIR}/headers/${_name}.cpp")
    endforeach()
endforeach()
add_library(minvex_sdk_headers OBJECT ${_header_checks})
target_link_libraries(minvex_sdk_headers PRIVATE minvex::core)

if(MINVEX_SDK_BUILD_VIEWER)
    add_executable(minvex_sdk_viewer Viewer.cpp)
    target_link_libraries(minvex_sdk_viewer PRIVATE minvex::viewer)
    add_executable(minvex_sdk_effects ../RenderingEnhancement/Main.cpp)
    target_link_libraries(minvex_sdk_effects PRIVATE minvex::viewer)
endif()
if(MINVEX_SDK_BUILD_EMBEDDED_HOST)
    if(WIN32)
        add_executable(minvex_sdk_host ../EmbeddedViewer/MainD3D11.cpp)
        target_compile_definitions(minvex_sdk_host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
        target_link_libraries(minvex_sdk_host PRIVATE minvex::embedded d3d11 dxgi user32)
    elseif(APPLE)
        enable_language(OBJCXX)
        add_executable(minvex_sdk_host ../EmbeddedViewer/MainMetal.mm)
        set_target_properties(minvex_sdk_host PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
        target_compile_options(minvex_sdk_host PRIVATE -fobjc-arc)
        target_link_libraries(minvex_sdk_host PRIVATE minvex::embedded "-framework Cocoa" "-framework Metal" "-framework QuartzCore")
    else()
        message(FATAL_ERROR "The native Host requires Windows or macOS.")
    endif()
    target_compile_definitions(minvex_sdk_host PRIVATE MINVEX_EMBEDDED_SAMPLE="${CMAKE_CURRENT_SOURCE_DIR}/../assets/AnimatedRibbon.gltf")
endif()
